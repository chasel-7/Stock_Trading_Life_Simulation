```typescript
/**
 * 生成样本行情数据的辅助脚本
 * 运行: npx ts-node scripts/generateMarketData.ts
 */
import * as fs from 'fs';

const SECTORS = ['消费', '科技', '制造', '医药', '金融'];
const STOCKS_PER_SECTOR = 3;
const TOTAL_DAYS = 15;
const TICKS_PER_DAY = 40;

interface TickData {
  open: number; close: number; high: number; low: number;
  ticks: number[];
}

function generateDayTicks(basePrice: number, volatility: number): TickData {
  const ticks: number[] = [];
  let price = basePrice;
  let high = price, low = price;

  for (let i = 0; i < TICKS_PER_DAY; i++) {
    const change = (Math.random() - 0.48) * volatility * basePrice;
    price = Math.max(price + change, basePrice * 0.9);
    price = Math.min(price, basePrice * 1.1);
    price = Math.round(price * 100) / 100;
    high = Math.max(high, price);
    low = Math.min(low, price);
    ticks.push(price);
  }

  return {
    open: basePrice,
    close: ticks[ticks.length - 1],
    high: Math.round(high * 100) / 100,
    low: Math.round(low * 100) / 100,
    ticks,
  };
}

function generateStockData(basePrice: number, volatility: number): TickData[] {
  const days: TickData[] = [];
  let currentPrice = basePrice;
  for (let d = 0; d < TOTAL_DAYS; d++) {
    const day = generateDayTicks(currentPrice, volatility);
    days.push(day);
    currentPrice = day.close;
  }
  return days;
}

const volatilityMap: Record<string, number> = {
  '消费': 0.015, '科技': 0.03, '制造': 0.02, '医药': 0.025, '金融': 0.01,
};

const stocks: Record<string, { dailyData: TickData[] }> = {};
for (const sector of SECTORS) {
  for (let i = 1; i <= STOCKS_PER_SECTOR; i++) {
    const id = `${sector}-0${i}`;
    const basePrice = 10 + Math.random() * 35;
    stocks[id] = {
      dailyData: generateStockData(
        Math.round(basePrice * 100) / 100,
        volatilityMap[sector]
      ),
    };
  }
}

const market = {
  marketId: 'sample-001',
  name: '策划行情-新手友好',
  totalDays: TOTAL_DAYS,
  recommendedStockIds: ['科技-01', '消费-02', '制造-03'],
  stocks,
};

fs.writeFileSync(
  'src/data/sampleMarket.json',
  JSON.stringify(market, null, 2)
);
console.log(`Generated market data: ${Object.keys(stocks).length} stocks × ${TOTAL_DAYS} days`);
```

```typescript
import { DailyStockData } from '../models/types';

interface MarketData {
  marketId: string;
  name: string;
  totalDays: number;
  recommendedStockIds: string[];
  stocks: Record<string, { dailyData: DailyStockData[] }>;
}

export class StockManager {
  private market: MarketData;
  private recommendedSet: Set<string>;

  constructor(marketData: MarketData) {
    this.market = marketData;
    this.recommendedSet = new Set(marketData.recommendedStockIds);
  }

  getStockIds(): string[] {
    return Object.keys(this.market.stocks);
  }

  getDailyData(stockId: string, day: number): DailyStockData {
    const stock = this.market.stocks[stockId];
    if (!stock) throw new Error(`Unknown stock: ${stockId}`);
    if (day < 0 || day >= stock.dailyData.length) {
      throw new Error(`Invalid day ${day} for stock ${stockId}`);
    }
    return stock.dailyData[day];
  }

  getTickPrice(stockId: string, day: number, tickIndex: number): number {
    const dailyData = this.getDailyData(stockId, day);
    const clampedIndex = Math.min(tickIndex, dailyData.ticks.length - 1);
    return dailyData.ticks[Math.max(0, clampedIndex)];
  }

  getClosePrice(stockId: string, day: number): number {
    return this.getDailyData(stockId, day).close;
  }

  isRecommended(stockId: string): boolean {
    return this.recommendedSet.has(stockId);
  }

  getTotalDays(): number {
    return this.market.totalDays;
  }
}
```

```typescript
import { GameState, Holding, DailySnapshot, MoodLevel } from '../models/types';
import { GAME_CONSTANTS } from '../config/constants';

export class GameStateManager {
  private state: GameState;

  constructor(characterId: string, startingCash: number, totalDays: number) {
    this.state = {
      currentDay: 1,
      totalDays,
      cash: startingCash,
      holdings: [],
      startingCash,
      characterId,
      phase: 'pre-market',
      dailySnapshots: [],
      collectedInfo: [],
      commissionDiscountDays: 0,
      gameOver: false,
      gameResult: 'playing',
    };
  }

  getState(): Readonly<GameState> {
    return { ...this.state, holdings: [...this.state.holdings] };
  }

  /** 买入股票 */
  buyStock(stockId: string, amount: number, price: number, commissionRate: number): void {
    const commission = Math.round(amount * commissionRate * 100) / 100;
    const totalCost = amount + commission;
    if (totalCost > this.state.cash) {
      throw new Error('Insufficient cash');
    }
    this.state.cash -= totalCost;
    const shares = amount / price;

    const existing = this.state.holdings.find(h => h.stockId === stockId);
    if (existing) {
      existing.costBasis += amount;
      existing.shares += shares;
    } else {
      this.state.holdings.push({ stockId, shares, costBasis: amount });
    }
  }

  /** 卖出股票 */
  sellStock(stockId: string, amount: number, price: number, commissionRate: number): void {
    const idx = this.state.holdings.findIndex(h => h.stockId === stockId);
    if (idx === -1) throw new Error(`No holding for ${stockId}`);

    const holding = this.state.holdings[idx];
    const sharesToSell = amount / price;
    if (sharesToSell > holding.shares + 0.001) {
      throw new Error('Insufficient shares');
    }

    const commission = Math.round(amount * commissionRate * 100) / 100;
    this.state.cash += amount - commission;

    holding.shares -= sharesToSell;
    holding.costBasis -= (holding.costBasis / (holding.shares + sharesToSell)) * sharesToSell;

    if (holding.shares <= 0.001) {
      this.state.holdings.splice(idx, 1);
    }
  }

  addCash(amount: number): void {
    this.state.cash += amount;
  }

  deductCash(amount: number): void {
    this.state.cash -= amount;
  }

  isBankrupt(): boolean {
    return this.state.cash <= 0;
  }

  /** 计算总资产 */
  getTotalAssets(currentPrices: Record<string, number>): number {
    let holdingsValue = 0;
    for (const h of this.state.holdings) {
      const price = currentPrices[h.stockId] || 0;
      holdingsValue += h.shares * price;
    }
    return Math.round((this.state.cash + holdingsValue) * 100) / 100;
  }

  isWin(currentPrices: Record<string, number>): boolean {
    return this.getTotalAssets(currentPrices) >= this.state.startingCash * GAME_CONSTANTS.WIN_MULTIPLIER;
  }

  isLastDay(): boolean {
    return this.state.currentDay >= this.state.totalDays;
  }

  /** 推进到下一天 */
  advanceDay(dailyReturn: number, mood: MoodLevel): void {
    this.state.dailySnapshots.push({
      day: this.state.currentDay,
      cash: this.state.cash,
      holdingsValue: 0, // 由调用方填充
      totalAssets: 0,
      dailyReturn,
      mood,
    });
    this.state.currentDay++;
    this.state.phase = 'pre-market';
  }

  setPhase(phase: GameState['phase']): void {
    this.state.phase = phase;
  }

  endGame(result: GameState['gameResult']): void {
    this.state.gameOver = true;
    this.state.gameResult = result;
  }
}
```
